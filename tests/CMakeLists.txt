set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

add_executable(test_core test_core.cpp)
target_link_libraries(test_core PRIVATE phy104_core)
add_test(NAME core_tests COMMAND test_core)

# Real JSON files + round-trip test of every formula (needs the Qt data layer)
if(TARGET phy104_data)
    add_executable(test_data test_data.cpp)
    target_link_libraries(test_data PRIVATE phy104_data phy104_core)
    target_compile_definitions(test_data PRIVATE DATA_DIR="${CMAKE_SOURCE_DIR}/data")
    add_test(NAME data_tests COMMAND test_data)
endif()

# UI-level checks: LaTeX renderer, SVG diagrams, theme, fonts, Markdown highlighter (offscreen, no window)
if(TARGET phy104_ui)
    find_package(Qt6 REQUIRED COMPONENTS Widgets Svg PrintSupport)
    add_executable(test_ui test_ui.cpp ${CMAKE_SOURCE_DIR}/assets/assets.qrc)
    target_link_libraries(test_ui PRIVATE phy104_ui)
    target_compile_definitions(test_ui PRIVATE
        DATA_DIR="${CMAKE_SOURCE_DIR}/data" ASSETS_DIR="${CMAKE_SOURCE_DIR}/assets")
    add_test(NAME ui_tests COMMAND test_ui)
    set_tests_properties(ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

# Clicks and key presses on the real application (offscreen)
if(TARGET phy104_ui)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test_interaction test_interaction.cpp ${CMAKE_SOURCE_DIR}/assets/assets.qrc)
    target_link_libraries(test_interaction PRIVATE phy104_ui Qt6::Test)
    target_compile_definitions(test_interaction PRIVATE DATA_DIR="${CMAKE_SOURCE_DIR}/data")
    add_test(NAME interaction_tests COMMAND test_interaction)
    set_tests_properties(interaction_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
endif()
